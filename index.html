import gradio as gr, soundfile as sf
from f5_tts_th.tts import TTS
tts = TTS(model="v1")
def run(a, r, g):
    w = tts.infer(ref_audio=a, ref_text=r, gen_text=g)
    sf.write("out.wav", w, 24000)
    return "out.wav"
gr.Interface(fn=run, inputs=[gr.Audio(type="filepath"), "text", "text"], outputs="audio").launch(share=True)
